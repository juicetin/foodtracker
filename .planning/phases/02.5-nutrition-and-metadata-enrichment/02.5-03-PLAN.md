    ```typescript
    // Import bundled JSON assets
    const nutritionMap = require('../../../assets/data/class-nutrition-map.json');
    const cuisineMap = require('../../../assets/data/cuisine-metadata.json');

    export interface NutritionMapping {
      fdcId: number | null;
      source: string;
      caloriesPer100g: number;
      proteinPer100g: number;
      carbsPer100g: number;
      fatPer100g: number;
      servingG: number;
      matchMethod: string;
    }

    export interface CuisineInfo {
      cuisine: string;
      country: string | null;
      region: string | null;
      translations: Record<string, string>;
    }

    export class ClassNutritionMapper {
      /** Get nutrition data for a classifier class name (normalized lowercase-hyphenated). */
      getNutritionMapping(className: string): NutritionMapping { ... }

      /** Get cuisine/cultural context for a classifier class name. Returns null if no cuisine data. */
      getCuisineInfo(className: string): CuisineInfo | null { ... }

      /** Convenience: get estimated macros for a given weight. */
      getMacrosForWeight(className: string, weightGrams: number): { calories: number; protein: number; carbs: number; fat: number } { ... }
    }
    ```
