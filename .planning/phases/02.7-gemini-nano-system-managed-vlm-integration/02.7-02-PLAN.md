```typescript
export async function runVlmIdentification(
  photoUri: string,
  items: DetectedItem[],
  userText?: string,
): Promise<DetectedItem[]> {
  // VLM is required — YOLO labels alone are not usable
  if (!vlmService.isReady) {                           // ← LINE 89: INSERT TIER 0 CHECK BEFORE THIS
    throw new Error('VLM model is not loaded. Download the VLM pack first.');
  }

  // Run VLM inference with retry
  const vlmResult = await identifyWithRetry(photoUri, userText);

  // If VLM returned no dishes, return items unchanged
  if (vlmResult.dishes.length === 0) {
    return items;
  }

  // Positional matching: assign dishes to items by bbox area descending
  const matchMap = matchVlmToItems(items, vlmResult);

  // Look up KG nutrition for matched VLM dishes
  const kgService = await getKnowledgeGraphService();

  // Build identified items
  const identified = await Promise.all(
    items.map(async (item) => {
      const vlmDish = matchMap.get(item.id);
      if (!vlmDish) return item;

      if (kgService) {
        try {
          await kgService.searchDish(vlmDish.name);
        } catch { /* KG lookup failed -- still apply VLM label */ }
      }

      return {
        ...item,
        vlmLabel: vlmDish.name,
        vlmCuisine: vlmDish.cuisine,
        vlmIngredients: vlmDish.ingredients,
        vlmConfidence: 0.8,
      };
    }),
  );

  return identified;
}
```

```typescript
// Internal: Gemini Nano identification path (Tier 0)

/**
 * Run food identification via Gemini Nano and apply the same downstream
 * positional matching + KG lookup as the SmolVLM path.
 *
 * Both paths return VlmFoodResult, so all downstream logic is shared.
 */
async function runGeminiNanoIdentification(
  photoUri: string,
  items: DetectedItem[],
  userText?: string,
): Promise<DetectedItem[]> {
  let vlmResult: VlmFoodResult;
  try {
    vlmResult = await geminiNanoService.identify(photoUri, userText);
  } catch (err) {
    // One silent retry on BUSY or transient error, then fall through to empty
    console.warn('[GeminiNano] First attempt failed:', err instanceof Error ? err.message : err);
    try {
      vlmResult = await geminiNanoService.identify(photoUri, userText);
    } catch (err2) {
      console.warn('[GeminiNano] Second attempt failed:', err2 instanceof Error ? err2.message : err2);
      vlmResult = { dishes: [] };
    }
  }

  if (vlmResult.dishes.length === 0) {
    return items;
  }

  // Reuse positional matching and KG lookup from the SmolVLM path
  const matchMap = matchVlmToItems(items, vlmResult);
  const kgService = await getKnowledgeGraphService();

  return Promise.all(
    items.map(async (item) => {
      const vlmDish = matchMap.get(item.id);
      if (!vlmDish) return item;

      if (kgService) {
        try {
          await kgService.searchDish(vlmDish.name);
        } catch { /* KG lookup failed -- still apply label */ }
      }

      return {
        ...item,
        vlmLabel: vlmDish.name,
        vlmCuisine: vlmDish.cuisine,
        vlmIngredients: vlmDish.ingredients,
        vlmConfidence: 0.85, // Gemini Nano confidence (slightly higher than SmolVLM default)
      };
    }),
  );
}
```
