def test_quantity_parsing_improved(kg_db):
    """At least 60% of recipe_ingredient rows should have non-50g quantities."""
    cursor = kg_db.cursor()
    cursor.execute("SELECT COUNT(*) FROM recipe_ingredient WHERE quantity_grams <> 50.0")
    non_default = cursor.fetchone()[0]
    cursor.execute("SELECT COUNT(*) FROM recipe_ingredient")
    total = cursor.fetchone()[0]
    pct = non_default / total * 100 if total > 0 else 0
    assert pct > 60, f"Expected >60% non-default quantities, got {pct:.1f}%"
```

def test_classifier_label_coverage(kg_db):
    """All named classifier labels should have KG dish entries."""
    import json
    labels_path = Path(__file__).parent.parent.parent / "apps" / "mobile" / "assets" / "models" / "labels_classify.json"
    if not labels_path.exists():
        pytest.skip("labels_classify.json not found")
    with open(labels_path) as f:
        data = json.load(f)
    named_labels = [l for l in data["labels"] if not l.isdigit() and not (len(l) == 3 and l[0] == '0')]

    cursor = kg_db.cursor()
    missing = []
    for label in named_labels:
        norm = label.lower().replace("-", " ").replace("_", " ").strip()
        cursor.execute("SELECT id FROM dish WHERE canonical_name = ?", (norm,))
        if not cursor.fetchone():
            missing.append(label)

    coverage_pct = (len(named_labels) - len(missing)) / len(named_labels) * 100
    assert coverage_pct > 95, f"Classifier label coverage: {coverage_pct:.1f}% (missing: {missing[:10]}...)"
```
